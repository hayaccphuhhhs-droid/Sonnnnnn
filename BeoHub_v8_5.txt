local function saveCfg()
    pcall(function()
        if not isfolder then return end
        if not isfolder(cf) then makefolder(cf) end
        local d = {}
        for k, v in pairs(getgenv()) do
            if (type(v) == "boolean" or type(v) == "number" or type(v) == "string")
                and k ~= "AntiAdmin" and k ~= "AdminIn" then
                d[k] = v
            end
        end
        writefile(cfPath, hs:JSONEncode(d))
    end)
end

local function makeDrag(frame, handle)
    local dragging, startMouse, startPos
    handle.InputBegan:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            startMouse = i.Position
            startPos = frame.Position
            i.Changed:Connect(function()
                if i.UserInputState == Enum.UserInputState.End then dragging = false end
            end)
        end
    end)
    uis.InputChanged:Connect(function(i)
        if not dragging then return end
        if i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch then
            local d = i.Position - startMouse
            frame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + d.X, startPos.Y.Scale, startPos.Y.Offset + d.Y)
        end
    end)
end

local opened, animating = false, false
local function toggleUI(v)
    if animating or v == opened then return end
    animating = true
    opened = v
    if opened then
        main.Visible = true
        scale.Scale = 0.88
        main.BackgroundTransparency = 0.35
        mainStroke.Transparency = 0.85
        mainStroke.Color = Color3.fromRGB(60, 255, 140)
        Tween(scale, 0.32, { Scale = 1 }, Enum.EasingStyle.Back, Enum.EasingDirection.Out)
        Tween(main, 0.28, { BackgroundTransparency = 0.04 })
        Tween(mainStroke, 0.3, { Transparency = 0.15, Color = Color3.fromRGB(60, 255, 140) })
        Tween(bStroke, 0.2, { Color = Color3.fromRGB(60, 255, 140), Transparency = 0.05, Thickness = 2.4 })
        Tween(bubble, 0.2, { TextColor3 = Color3.fromRGB(60, 255, 140) })
        bubble.Text = "x"
        task.delay(0.35, function() animating = false end)
    else
        Tween(scale, 0.22, { Scale = 0.9 }, Enum.EasingStyle.Quint, Enum.EasingDirection.In)
        Tween(main, 0.22, { BackgroundTransparency = 0.4 }, Enum.EasingStyle.Quint, Enum.EasingDirection.In)
        Tween(mainStroke, 0.2, { Transparency = 0.55, Color = Color3.fromRGB(0, 200, 245) })
        Tween(bStroke, 0.2, { Color = ac, Transparency = 0.2, Thickness = 2 })
        Tween(bubble, 0.2, { TextColor3 = ac })
        bubble.Text = "B"
        task.delay(0.24, function()
            if not opened then main.Visible = false end
            animating = false
        end)
    end
end

        local active = n == name

        end
    end

end

    hit.Position = UDim2.new(0, 0, -1, 0)
    hit.BackgroundTransparency = 1
    hit.Text = ""
    hit.MouseButton1Down:Connect(function()
        local dragging = true
        local c1, c2
        c1 = uis.InputChanged:Connect(function(i)
            if not dragging then return end
            if i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch then
                local rel = math.clamp((i.Position.X - bar.AbsolutePosition.X) / math.max(bar.AbsoluteSize.X, 1), 0, 1)
                local step = (max - min) <= 5 and 0.05 or 1
                local val = min + (max - min) * rel
                if step >= 1 then val = math.floor(val + 0.5)
                else val = math.floor(val / step + 0.5) * step; val = math.floor(val * 100 + 0.5) / 100 end
                fill.Size = UDim2.new(rel, 0, 1, 0)
                l.Text = t .. " : " .. tostring(val)
                set(val)
                saveCfg()
            end
        end)
        c2 = uis.InputEnded:Connect(function(i)
            if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
                dragging = false
                if c1 then c1:Disconnect() end
                if c2 then c2:Disconnect() end
            end
        end)
    end)
end

local function getBuyableBaitList()
    local list, seen = {}, {}
    local info = rep:FindFirstChild("Info")
    local baitFolder = info and info:FindFirstChild("Bait")
    if baitFolder then
        for _, c in ipairs(baitFolder:GetChildren()) do
            if c:IsA("ModuleScript") then
                local ok, data = pcall(require, c)
                if ok and type(data) == "table" and not data.Ingredient and not seen[c.Name] then
                    seen[c.Name] = true
                    table.insert(list, c.Name)
                end
            end
        end
    end
    local itemFolder = info and info:FindFirstChild("Item")
    if itemFolder then
        for _, c in ipairs(itemFolder:GetChildren()) do
            if c:IsA("ModuleScript") and not seen[c.Name] then
                local ok, data = pcall(require, c)
                if ok and type(data) == "table" and data.Type == "Bait" and not data.Ingredient then
                    seen[c.Name] = true
                    table.insert(list, c.Name)
                end
            end
        end
    end
    table.sort(list)
    return list
end

local oW, wS = {}, false
function setWalkOnWater(on)
    if on then
        for _, v in pairs(workspace:GetDescendants()) do
            if v:IsA("BasePart") and v.Name:lower() == "water" then
                if oW[v] == nil then oW[v] = v.CanCollide end
                v.CanCollide = true
            end
        end
        local ocean = workspace:FindFirstChild("Ocean")
        if ocean then
            for _, waterPart in ipairs(ocean:GetChildren()) do
                if waterPart.Name == "Water" and waterPart:IsA("BasePart") then
                    if oW[waterPart] == nil then oW[waterPart] = waterPart.CanCollide end
                    waterPart.CanCollide = true
                end
            end
        end
    else
        for w, o in pairs(oW) do
            if w and w.Parent then w.CanCollide = o end
        end
        table.clear(oW)
    end
    wS = on and true or false
end

                    if waterPart.Name == "Water" and waterPart:IsA("BasePart") and not waterPart.CanCollide then
                        if oW[waterPart] == nil then oW[waterPart] = false end
                        waterPart.CanCollide = true
                    end
                end
            end
        end
    end
end)

local lastDailyQuestClock, lastDailyRewardClock = 0, 0
local function stepAutoClaimDailyQuest()
    if not getgenv().AutoClaimDaily or getgenv().AdminIn then return end
    local now = os.clock()
    if now - lastDailyQuestClock < 2 then return end
    lastDailyQuestClock = now
    pcall(function()
        local pData = getPlayerData()
        local questFolder = pData and pData:FindFirstChild("Quest")
        local dailyFolder = questFolder and questFolder:FindFirstChild("Daily")
        if not dailyFolder then return end
        local questInfoTable = nil
        local questInfoMod = InfoFolder and InfoFolder:FindFirstChild("Quest")
        if questInfoMod then pcall(function() questInfoTable = require(questInfoMod) end) end
        local claimEv = evs:FindFirstChild("ClaimQuest")
        if not claimEv then return end
        for _, dailyEntry in ipairs(dailyFolder:GetChildren()) do
            local claimVal = dailyEntry:FindFirstChild("Claim")
            local progVal = dailyEntry:FindFirstChild("Progression")
            local questTypeVal = dailyEntry:FindFirstChild("QuestType")
            if claimVal and progVal and questTypeVal and claimVal.Value == false and questTypeVal.Value ~= "" then
                local reqAmount = nil
                if questInfoTable and questInfoTable.Daily and tonumber(dailyEntry.Name) then
                    local dailyDef = questInfoTable.Daily[tonumber(dailyEntry.Name)]
                    if dailyDef and dailyDef[2] then reqAmount = dailyDef[2] end
                end
                if reqAmount then
                    local curProg = progVal.Value
                    if questTypeVal.Value == "PlayTime" then
                        if math.floor(curProg / 60) >= math.floor(reqAmount / 60) then
                            claimEv:FireServer(dailyEntry.Name)
                        end
                    else
                        if curProg >= reqAmount then
                            claimEv:FireServer(dailyEntry.Name)
                        end
                    end
                end
            end
        end
    end)
end

local function stepAutoClaimDailyReward()
    if not getgenv().AutoDailyReward or getgenv().AdminIn then return end
    local now = os.clock()
    if now - lastDailyRewardClock < 5 then return end
    lastDailyRewardClock = now
    pcall(function()
        local pData = getPlayerData()
        local dailyRewardFolder = pData and pData:FindFirstChild("DailyReward")
        if not dailyRewardFolder then return end
        local dayCycleVal = dailyRewardFolder:FindFirstChild("DayCycle")
        local dayCycleStr = dayCycleVal and dayCycleVal.Value
        if not dayCycleStr or dayCycleStr == "" then return end
        local currentDayIndex = math.floor(workspace:GetServerTimeNow() / 86400)
        local lastClaimedDay = tonumber(dayCycleStr)
        if lastClaimedDay and lastClaimedDay >= currentDayIndex then return end
        local dailyRewardEvent = evs:FindFirstChild("DailyReward")
        if not dailyRewardEvent then return end
        for dayNum = 1, 7 do
            local dayVal = dailyRewardFolder:FindFirstChild(tostring(dayNum))
            if dayVal and dayVal.Value == false then
                local ok = dayNum == 1
                if not ok then
                    local prev = dailyRewardFolder:FindFirstChild(tostring(dayNum - 1))
                    ok = prev and prev.Value == true
                end
                if ok then
                    dailyRewardEvent:FireServer(dayNum)
                    break
                end
            end
        end
    end)
end

        if inst:IsA("NumberValue") or inst:IsA("IntValue") then
            n = tonumber(inst.Value) or 0
        else
            local q = inst:FindFirstChild("Quantity") or inst:FindFirstChild("Amount") or inst:FindFirstChild("Count")
            if q and q:IsA("ValueBase") then
                n = tonumber(q.Value) or 0
            else
                n = 1
            end
        end
    end)
    return math.max(0, n)
end

local function getBasicBaitCount()
    local pData = getPlayerData()
    if not pData then return 0 end
    local total = 0
    local function scan(folder)
        if not folder then return end
        for _, child in ipairs(folder:GetChildren()) do
            local n = cleanBaitName(child.Name):lower()
            if n == "basic bait" then
                total = total + getBaitItemCount(child)
            end
        end

    scan(pData:FindFirstChild("Inventory"))

local function getEquippedBaitQuantity()

local function isBasicEquippedNow()
    if equipLocked and os.clock() < equipLockUntil then
        return true
    end
    local name = getEquippedBaitName()
    local qty = getEquippedBaitQuantity()
    if name and name:lower() == "basic bait" and qty > 0 then
        equipLocked = true
        equipLockUntil = os.clock() + 60
        return true
    end
    -- nếu đọc được bait KHÁC rõ ràng → mở khóa để equip lại
    if name and name:lower() ~= "basic bait" and qty > 0 then
        equipLocked = false
        return false
    end
    if qty == 0 and name == nil then
        -- không chắc: nếu đang lock thì coi như vẫn ok
        return equipLocked
    end
    return equipLocked
end

-- CHỈ gọi remote khi chưa basic; SAU KHI gọi → lock, KHÔNG recheck Fire lần 2
function StepAutoEquipBasicBait(forceNotify)
    if not getgenv().AutoEquipBasicBait and not forceNotify then
        return
    end

    -- đã lock / đã basic → tuyệt đối không gửi (tránh toggle tháo)
    if not forceNotify and isBasicEquippedNow() then
        return
    end

    local now = os.clock()
    if not forceNotify and (now - lastEquipBaitClock) < 5 then
        return
    end

    if getBasicBaitCount() <= 0 then
        return
    end

    -- force từ nút UI: cho phép 1 lần, vẫn lock sau đó
    lastEquipBaitClock = now
    if fireEquipBaitOnce() then
        equipLocked = true
        equipLockUntil = now + 60 -- 60s không đụng remote equip
        -- KHÔNG task.delay re-fire (đó là nguyên nhân toggle tháo)
    end
end

local function isDailyTicketLimitText(msg)
    if not msg or msg == "" then return false end
    local m = string.lower(msg)
    return m:find("daily limit", 1, true) and (m:find("20", 1, true) or m:find("come back tomorrow", 1, true))
end

-- "another quest in 20:00" / "15m" / "30s"
local function parseQuestCooldownSeconds(msgText)
    if not msgText or msgText == "" then return nil end
    local lowerMsg = string.lower(msgText)
    if not (string.find(lowerMsg, "another quest in", 1, true)
        or string.find(lowerMsg, "take another quest in", 1, true)
        or string.find(lowerMsg, "wait", 1, true) and string.find(lowerMsg, "quest", 1, true)) then
        return nil
    end
    local minsStr, secsStr = string.match(lowerMsg, "(%d+)%s*:%s*(%d+)")
    if minsStr and secsStr then
        return (tonumber(minsStr) or 0) * 60 + (tonumber(secsStr) or 0)
    end
    local minMatch = string.match(lowerMsg, "(%d+)%s*m")
    local secMatch = string.match(lowerMsg, "(%d+)%s*s")
    if minMatch or secMatch then
        return (tonumber(minMatch) or 0) * 60 + (tonumber(secMatch) or 0)
    end
    return nil
end

local function inspectTicketMessage(msgText)
    if not getgenv().AutoTicketQuest or not msgText or msgText == "" or isTicketLimitReached then return end
    if #msgText > 300 then return end
    if isDailyTicketLimitText(msgText) then
        isTicketLimitReached = true
        getgenv().AutoTicketQuest = false
        pcall(function()
            sg:SetCore("SendNotification", {
                Title = "Ticket",
                Text = "Daily limit reached",
                Duration = 4,
            })
        end)
        return
    end
    if msgText == lastCooldownMsg then return end
    local sec = parseQuestCooldownSeconds(msgText)
    if sec and sec > 0 and sec <= 3600 then
        lastCooldownMsg = msgText
        ticketCooldownUntil = os.clock() + sec
        pcall(function()
            sg:SetCore("SendNotification", {
                Title = "Ticket CD",
                Text = string.format("Wait %dm %ds", math.floor(sec / 60), sec % 60),
                Duration = 3,
            })
        end)
    end
end

            local hbR = hbL + hbW
            if barC >= hbL and barC <= hbR then
                lastMgHit = now
                local mob = fishing:FindFirstChild("MobileFishing", true) or fishing:FindFirstChild("Button", true)
                if mob then clickGui(mob) end
                pcall(function()
                    if evs:FindFirstChild("UpdateFishProgression") then
                        evs.UpdateFishProgression:FireServer()
                    end
                end)
            end
        end
    end

    -- rhythm ASD
    local rhythm = fishing:FindFirstChild("Rhythm", true) or fishing:FindFirstChild("RhythmGui", true)
    if rhythm and rhythm.Visible then
        for _, laneKey in ipairs({"A", "S", "D"}) do
            if now - (lastRhythmLane[laneKey] or 0) >= math.max(0.05, delay) then
                local lane = rhythm:FindFirstChild("Progression" .. laneKey)
                if lane then
                    local barFrame = lane:FindFirstChild("BarFrame")
                    local targetY = barFrame and barFrame.Position.Y.Scale or 0.85
                    local best, bestDist = nil, 0.16
                    for _, ch in ipairs(lane:GetChildren()) do
                        if ch.Name == "Note_FX" and ch:IsA("GuiObject") and ch.Visible then
                            local d = math.abs(ch.Position.Y.Scale - targetY)
                            if d < bestDist then bestDist, best = d, ch end
                        end
                    end
                    if best then
                        lastRhythmLane[laneKey] = now
                        local laneBtn = lane:FindFirstChild("Button")
                        if laneBtn then clickGui(laneBtn) end
                        local kc = rhythmKeyMap[laneKey]
                        if kc then
                            pcall(function()
                                vim:SendKeyEvent(true, kc, false, game)
                                task.delay(0.02, function()
                                    vim:SendKeyEvent(false, kc, false, game)
                                end)
                            end)
                        end
                    end
                end
            end
        end
    end
end

local function stepAutoBoss()
    if getgenv().AdminIn then return end
    local wantEnzo = getgenv().AutoEnzo
    local wantLong = getgenv().AutoLong
    if not wantEnzo and not wantLong then return end
    local now = os.clock()
    if now - lastBossStep < 1.2 then return end
    lastBossStep = now

    local char = me.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    if not hrp then return end

    -- if fishing minigame open, let minigame handler work
    pcall(refreshGui)
    if FishingGui and FishingGui.Visible then
        return
    end

    local targetCF = wantLong and BossCF.Long or BossCF.Enzo
    if (hrp.Position - targetCF.Position).Magnitude > 18 then
        hrp.CFrame = targetCF
        return
    end

    -- try open boss menu / fight
    local pg = me:FindFirstChild("PlayerGui")
    local mainGui = pg and pg:FindFirstChild("MainGui")
    if not mainGui then return end
    local menu = mainGui:FindFirstChild("Menu")
    local modeWant = string.lower(tostring(getgenv().BossMode or "Normal"))

    -- click Fight / mode buttons if visible
    pcall(function()
        for _, f in ipairs(mainGui:GetDescendants()) do
            if f:IsA("TextButton") and f.Visible then
                local t = string.lower(f.Text or "")
                local title = f:FindFirstChild("Title")
                if title and title:IsA("TextLabel") then
                    t = string.lower(title.Text or t)
                end
                if string.find(t, modeWant, 1, true)
                    or string.find(t, "fight", 1, true)
                    or string.find(t, "challenge", 1, true) then
                    if not string.find(t, "close") and not string.find(t, "cancel") then
                        clickGui(f)
                        break
                    end
                end
            end
        end
    end)

    -- proximity prompt / click detector on boss model
    pcall(function()
        local names = wantLong and {"Long", "DiaoHai", "Long DiaoHai"} or {"Enzo"}
        for _, m in ipairs(workspace:GetDescendants()) do
            if m:IsA("Model") then
                local n = m.Name
                for _, want in ipairs(names) do
                    if string.find(n, want, 1, true) then
                        local pp = m:FindFirstChildWhichIsA("ProximityPrompt", true)
                        if pp and fireproximityprompt then
                            fireproximityprompt(pp)
                        end
                    end
                end
            end
        end
    end)
end

task.spawn(function()
    while task.wait(0.35) do
        pcall(stepAutoBoss)
    end
end)

function unloadHub()
    killAll()
    pcall(function() setWalkOnWater(false) end)
    getgenv().FullBright = false
    getgenv().NoFog = false
    applyLighting()
    clrE()
    for p, d in pairs(getgenv().ESPPlayerList) do
        pcall(function() if d.hl then d.hl:Destroy() end end)
        pcall(function() if d.bill then d.bill:Destroy() end end)
        getgenv().ESPPlayerList[p] = nil
    end
    pcall(function()
        for _, g in ipairs({ game.CoreGui, me.PlayerGui }) do
            if g then
                local u = g:FindFirstChild("BeoHubUI")
                if u then u:Destroy() end
                local f = g:FindFirstChild("BeoFishBtn")
                if f then f:Destroy() end
                local e = g:FindFirstChild("BeoHubESPCards")
                if e then e:Destroy() end
            end
        end
    end)
    pcall(function()
        sg:SetCore("SendNotification", { Title = "BeoHub", Text = "Unloaded", Duration = 3 })
    end)
end
